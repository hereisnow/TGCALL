function connectGoogle() { const tg = window.Telegram.WebApp; const userId = tg.initDataUnsafe?.user?.id; if (!userId) { alert("Пожалуйста, откройте приложение через Telegram"); return; } const backendUrl = "https://tgcall-production.up.railway.app"; // Перенаправляем на авторизацию с передачей user_id window.location.href = `${backendUrl}/auth/google/start?user_id=${userId}`; }
